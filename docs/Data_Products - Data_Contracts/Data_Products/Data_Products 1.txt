📡 Produto A: Dashboard de Qualidade de Rede
Este produto foca-se na infraestrutura.
Grão : 1 linha por Antena (Torre) por Dia.

# ==============================================================================
# DATA PRODUCT: Qualidade de Rede Diária (gold.network_quality_daily)
# ==============================================================================
product:
  id: gold.network_quality_daily
  layer: gold
  domain: network_operations      # Domínio de negócio responsável
  owner:
    team: noc-engineering         # NOC - Network Operations Center
    contact: noc@telecom.pt
  description: "Métricas diárias de condições de rádio, retenção de chamadas e estado físico das infraestruturas." 
  consumers: # Quem vai consumir estes dados
    - name: bi.infrastructure_map # Dashboard geoespacial de Leiria
      type: bi
      access: trino_sql           # Vão aceder via SQL no Trino

# Como o dado é servido
delivery:
  storage:
    system: minio
    format: iceberg             # Formato nativo Lakehouse para tabelas fiáveis
    bucket: warehouse
    table_identifier: warehouse.gold.network_quality_daily
    partitions:
      - date                    # Particionado por data para queries eficientes
    update_mode: merge          # Upsert idempotente (evita duplicados se o Flyte correr 2x)
  serving:
    query_engine: trino
    stable_views:
      - warehouse.semantic.vw_network_quality_current

# Como o dado é gerado
production:
  orchestrator: flyte
  workflow: workflows.gold_network_quality_wf
  schedule: "0 3 * * *"         # Corre todos os dias às 03:00 da manhã
  backfills:
    supported: true             # Se descobrirmos um erro, podemos recalcular o passado
    unit: day

# De onde vêm os dados (Linhagem)
lineage:
  sources:
    - warehouse.silver.network_logs_clean
    - warehouse.silver.towers_status
    - warehouse.silver.call_tests_clean

# Observabilidade
observability:
  alerts:
    - type: freshness_breach
      notify: "#noc-data-alerts" # Alerta no Slack/Teams se os dados não chegarem a horas

references:
  contract_file: contracts/gold.network_quality_daily.yaml
