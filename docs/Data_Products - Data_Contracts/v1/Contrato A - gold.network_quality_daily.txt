  id: gold.network_quality_daily

  grain: "1 linha por (id_antena_conectada, dia de calendário)"

    nullable: true

  - name: latitude_ocorrencia

    nullable: true
  - name: longitude_ocorrencia
    type: double
    nullable: true
  - name: torre_latitude
    type: double
    nullable: true
  - name: torre_longitude
    type: double
    nullable: true
  - name: id_antena_conectada

    nullable: false

  - name: estado_antena

    nullable: true
    description: "status da torre no snapshot do dia."
  - name: distancia_antena_m
    type: double
    nullable: true

  definitions:
